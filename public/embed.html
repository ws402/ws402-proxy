How to Embed:

<!-- Add this to any HTML page -->
<div id="my-video"></div>

<script src="https://your-domain.com/embed.js"
        data-resource="abc123"
        data-target="my-video">
</script>
// React Component
import { useEffect } from 'react';

function VideoPlayer() {
  useEffect(() => {
    const script = document.createElement('script');
    script.src = 'https://your-domain.com/embed.js';
    script.setAttribute('data-resource', 'abc123');
    script.setAttribute('data-target', 'video-player');
    document.body.appendChild(script);
  }, []);

  return <div id="video-player" />;
}
<!-- Vue Component -->
<template>
  <div id="video-player"></div>
</template>

<script>
export default {
  mounted() {
    const script = document.createElement('script');
    script.src = 'https://your-domain.com/embed.js';
    script.setAttribute('data-resource', 'abc123');
    script.setAttribute('data-target', 'video-player');
    document.body.appendChild(script);
  }
}
</script>
